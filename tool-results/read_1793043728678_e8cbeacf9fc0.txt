    49→// ── Liquid-glass indicator ────────────────────────────────────────────
    50→//
    51→// The active capsule is driven by measured element positions — never by
    52→// hardcoded coordinates — so it stays glued to the selected item across
    53→// resizes, orientation changes, font loading and route changes. Movement is
    54→// transform-only (x/y + springs); velocity-derived scaleX/scaleY make the
    55→// capsule subtly stretch along its travel axis, then settle back to exactly
    56→// 1 (a scale factor must rest at 1 — resting at 0 would collapse the capsule
    57→// to zero width). Animation state lives entirely inside this hook, so section
    58→// switches never rerender page content.
    59→
    60→interface Rect {
    61→  x: number;
    62→  y: number;
    63→  w: number;
    64→  h: number;
    65→}
    66→
    67→const SPRING = { stiffness: 480, damping: 38, mass: 0.9 } as const;
    68→const STRETCH_SPRING = { stiffness: 420, damping: 40 } as const;
    69→
    70→/** Pointer travel needed before a press becomes a capsule drag (px). */
    71→const DRAG_THRESHOLD = 8;
    72→/** Releases up to this far outside the bar still count as a drop inside (px). */
    73→const DROP_MARGIN = 40;
    74→
    75→/** Peak travel velocity adds up to 14% extra scale along the movement axis. */
    76→const stretchOf = (v: number) => 1 + Math.min(0.14, Math.abs(v) / 3200);
    77→
    78→/** Placement must land before paint (no late pop-in / stale-frame flash);
    79→ *  useLayoutEffect does not exist on the server, so fall back to useEffect. */
    80→const useIsomorphicLayoutEffect =
    81→  typeof window !== 'undefined' ? useLayoutEffect : useEffect;
    82→
    83→interface DragGesture {
    84→  pointerId: number;
    85→  startX: number;
    86→  startY: number;
    87→  /** Grabbed point inside the capsule, in ul content coordinates. */
    88→  grabOffset: number;
    89→  /** Last pointer-derived capsule position along the drag axis. */
    90→  lastPos: number;
    91→  /** True once the press crossed the drag threshold. */
    92→  dragging: boolean;
    93→  ul: HTMLUListElement;
    94→  stop: () => void;
    95→}
    96→
    97→function useLiquidNav(
    98→  count: number,
    99→  activeIndex: number,
   100→  axis: 'x' | 'y',
   101→  onActivate?: (id: Section) => void,
   102→) {
   103→  const itemRefs = useRef<Array<HTMLElement | null>>([]);
   104→  const rectsRef = useRef<Array<Rect | null>>([]);
   105→  const gestureRef = useRef<DragGesture | null>(null);
   106→  /** Set after a real drag ends — swallows the leftover pointer click once. */
   107→  const swallowClickRef = useRef(false);
   108→  const placedRef = useRef(false);
   109→  const reduced = useReducedMotion();
   110→  const [rect, setRect] = useState<Rect | null>(null);
   111→
   112→  const measure = useCallback(() => {
   113→    // Every item's box, in ul content coordinates (offset* ignores transforms,
   114→    // so the capsule's own position can never skew the numbers).
   115→    rectsRef.current = itemRefs.current.map((el) =>
   116→      el && !(el.offsetWidth === 0 && el.offsetHeight === 0)
   117→        ? { x: el.offsetLeft, y: el.offsetTop, w: el.offsetWidth, h: el.offsetHeight }
   118→        : null,
   119→    );
   120→    console.debug('[dfnav] measure', axis, 'activeIndex', activeIndex, 'rect', rectsRef.current[activeIndex], 'all', rectsRef.current.map((r) => r && [r.x, r.y, r.w, r.h]));
   121→    setRect(rectsRef.current[activeIndex] ?? null);
   122→  }, [activeIndex]);
   123→
   124→  useIsomorphicLayoutEffect(() => {
   125→    if (activeIndex < 0) {
   126→      setRect(null);
   127→      return;
   128→    }
   129→    measure();
   130→    const el = itemRefs.current[activeIndex];
   131→    const container = (el?.offsetParent as HTMLElement | null) ?? null;
   132→    if (typeof ResizeObserver === 'undefined') {
   133→      window.addEventListener('resize', measure);
   134→      return () => window.removeEventListener('resize', measure);
   135→    }
   136→    const ro = new ResizeObserver(measure);
   137→    if (el) ro.observe(el);
   138→    if (container) ro.observe(container);
   139→    // Font swap changes item widths — re-measure once fonts settle.
   140→    const fonts = (document as Document & { fonts?: { ready: Promise<unknown> } }).fonts;
   141→    fonts?.ready.then(() => measure()).catch(() => {});
   142→    window.addEventListener('resize', measure);
   143→    window.addEventListener('orientationchange', measure);
   144→    return () => {
   145→      ro.disconnect();
   146→      window.removeEventListener('resize', measure);
   147→      window.removeEventListener('orientationchange', measure);
   148→    };
   149→  }, [measure, activeIndex, count]);
   150→
   151→  const x = useSpring(0, SPRING);
   152→  const y = useSpring(0, SPRING);
   153→  const w = useSpring(0, SPRING);
   154→  const h = useSpring(0, SPRING);
   155→
   156→  useIsomorphicLayoutEffect(() => {
   157→    console.debug('[dfnav] apply', axis, 'rect', rect && [rect.x, rect.y, rect.w, rect.h], 'dragging', !!gestureRef.current?.dragging, 'placed', placedRef.current, 'reduced', reduced);
   158→    if (gestureRef.current?.dragging) return; // never fight an active drag
   159→    if (!rect) {
   160→      // Capsule hidden — the next valid rect is a fresh placement, not a
   161→      // travel animation from stale coordinates.
   162→      placedRef.current = false;
   163→      return;
   164→    }
   165→    if (reduced || !placedRef.current) {
   166→      // First placement (or reduced motion): no travel animation.
   167→      x.jump(rect.x);
   168→      y.jump(rect.y);
   169→      w.jump(rect.w);
   170→      h.jump(rect.h);
   171→      placedRef.current = true;
   172→    } else {
   173→      x.set(rect.x);
   174→      y.set(rect.y);
   175→      w.set(rect.w);
   176→      h.set(rect.h);
   177→    }
   178→  }, [rect, reduced, x, y, w, h]);
   179→
   180→  // Stretch along the movement axis while travelling, settle to exactly 1.
   181→  // Per-axis: the dock travels vertically, the tab bar horizontally.
   182→  const stretchX = useSpring(useTransform(useVelocity(x), stretchOf), STRETCH_SPRING);
   183→  const stretchY = useSpring(useTransform(useVelocity(y), stretchOf), STRETCH_SPRING);
   184→
   185→  /** Spring the capsule into an item's exact bounds; jumps under reduced motion. */
   186→  const placeAt = useCallback(
   187→    (r: Rect) => {
   188→      if (reduced) {
   189→        x.jump(r.x);
   190→        y.jump(r.y);
   191→        w.jump(r.w);
   192→        h.jump(r.h);
   193→      } else {
   194→        x.set(r.x);
   195→        y.set(r.y);
   196→        w.set(r.w);
   197→        h.set(r.h);
   198→      }
   199→    },
   200→    [reduced, x, y, w, h],
   201→  );
   202→
   203→  /** End the gesture; a dragged capsule springs back to the active item. */
   204→  const abortGesture = useCallback(() => {
   205→    const gesture = gestureRef.current;
   206→    if (!gesture) return;
   207→    const wasDragging = gesture.dragging;
   208→    gestureRef.current = null;
   209→    gesture.stop();
   210→    if (wasDragging) {
   211→      const r = rectsRef.current[activeIndex];
   212→      if (r) placeAt(r);
   213→    }
   214→  }, [activeIndex, placeAt]);
   215→
   216→  // ── Draggable capsule ─────────────────────────────────────────────
   217→  //
   218→  // Press the ACTIVE item and move past a small threshold to grab the glass
   219→  // capsule; it then follows the pointer 1:1 (MotionValue writes only — no
   220→  // React renders per move, no spring lag during direct manipulation) until
   221→  // release. Releasing over/near the bar drops it on the nearest item and
   222→  // navigates there; releasing further out, losing the capture or cancelling
   223→  // returns it to the current route without navigating. Sub-threshold presses
   224→  // stay ordinary clicks, so clicking and keyboard behaviour are untouched.
   225→
   226→  const onPointerDown = useCallback(
   227→    (e: React.PointerEvent<HTMLUListElement>) => {
   228→      if (e.pointerType === 'mouse' && e.button !== 0) return;
   229→      if (gestureRef.current) return; // a gesture already owns this pointer
   230→      swallowClickRef.current = false;
   231→      const ul = e.currentTarget;
   232→      const btn = (e.target as HTMLElement).closest?.('button[data-nav-item]');
   233→      // Only the active item's capsule is grabbable — every other tab keeps
   234→      // its plain click behaviour.
   235→      if (!btn || btn !== itemRefs.current[activeIndex]) return;
   236→      const active = rectsRef.current[activeIndex];
   237→      if (!active) return;
   238→
   239→      const box = ul.getBoundingClientRect();
   240→      const pointerLocal =
   241→        axis === 'x'
   242→          ? e.clientX - box.left - ul.clientLeft + ul.scrollLeft
   243→          : e.clientY - box.top - ul.clientTop + ul.scrollTop;
   244→      const grabOffset = pointerLocal - (axis === 'x' ? active.x : active.y);
   245→
   246→      let gesture: DragGesture;
   247→
   248→      const onMove = (ev: PointerEvent) => {
   249→        if (ev.pointerId !== gesture.pointerId) return;
   250→        if (!gesture.dragging) {
   251→          if (
   252→            Math.hypot(ev.clientX - gesture.startX, ev.clientY - gesture.startY) <
   253→            DRAG_THRESHOLD
   254→          )
   255→            return;
   256→          gesture.dragging = true;
   257→          try {
   258→            // Keep receiving events even when the pointer leaves the capsule.
   259→            gesture.ul.setPointerCapture(gesture.pointerId);
   260→          } catch {
   261→            /* window listeners below keep the drag alive regardless */
   262→          }
   263→        }
   264→        ev.preventDefault(); // no text selection / native drag mid-gesture
   265→        const next = gesture.ul.getBoundingClientRect();
   266→        const raw =
   267→          axis === 'x'
   268→            ? ev.clientX -
   269→              next.left -
   270→              gesture.ul.clientLeft +
   271→              gesture.ul.scrollLeft -
   272→              gesture.grabOffset
   273→            : ev.clientY -
   274→              next.top -
   275→              gesture.ul.clientTop +
   276→              gesture.ul.scrollTop -
   277→              gesture.grabOffset;
   278→        // Clamp to the item extent so the capsule never leaves the bar.
   279→        const rects = rectsRef.current.filter(Boolean) as Rect[];
   280→        const pos =
   281→          rects.length === 0
   282→            ? raw
   283→            : Math.min(
   284→                Math.max(...rects.map((r) => (axis === 'x' ? r.x + r.w : r.y + r.h))) -
   285→                  (axis === 'x' ? active.w : active.h),
   286→                Math.max(
   287→                  Math.min(...rects.map((r) => (axis === 'x' ? r.x : r.y))),
   288→                  raw,
   289→                ),
   290→              );
   291→        gesture.lastPos = pos;
   292→        // 1:1 direct manipulation — jump(), so the rendering spring can never
   293→        // lag the finger; the release placement below does the springing.
   294→        if (axis === 'x') x.jump(pos);
   295→        else y.jump(pos);
   296→      };
   297→
   298→      const onUp = (ev: PointerEvent) => {
   299→        if (ev.pointerId !== gesture.pointerId) return;
   300→        const wasDragging = gesture.dragging;
   301→        gestureRef.current = null;
   302→        gesture.stop();
   303→        if (!wasDragging) return; // plain click — the button's onClick handles it
   304→        swallowClickRef.current = true; // the release click is ours to own
   305→
   306→        // Drop test: the pointer must be over (or near) the bar; anything
   307→        // further out counts as a cancelled drop and returns the capsule to
   308→        // the active item without navigating.
   309→        const drop = gesture.ul.getBoundingClientRect();
   310→        const insideBar =
   311→          ev.clientX >= drop.left - DROP_MARGIN &&
   312→          ev.clientX <= drop.right + DROP_MARGIN &&
   313→          ev.clientY >= drop.top - DROP_MARGIN &&
   314→          ev.clientY <= drop.bottom + DROP_MARGIN;
   315→
   316→        // Decide from the capsule's logical position (last pointer-derived
   317→        // value), never from the rendering spring — it can trail the pointer.
   318→        const capsuleCenter =
   319→          axis === 'x' ? gesture.lastPos + active.w / 2 : gesture.lastPos + active.h / 2;
   320→        let best = -1;
   321→        let bestDist = Number.POSITIVE_INFINITY;
   322→        rectsRef.current.forEach((r, i) => {
   323→          if (!r) return;
   324→          const center = axis === 'x' ? r.x + r.w / 2 : r.y + r.h / 2;
   325→          const dist = Math.abs(center - capsuleCenter);
   326→          if (dist < bestDist) {
   327→            bestDist = dist;
   328→            best = i;
   329→          }
   330→        });
   331→
   332→        const target = best >= 0 ? rectsRef.current[best] : null;
   333→        const targetId =
   334→          best >= 0
   335→            ? (itemRefs.current[best]?.getAttribute('data-nav-id') as Section | null)
   336→            : null;
   337→        if (insideBar && target && targetId) {
   338→          placeAt(target); // settle exactly into the item's bounds
   339→          if (targetId !== NAV_ITEMS[activeIndex]?.id) onActivate?.(targetId);
   340→        } else {
   341→          const home = rectsRef.current[activeIndex];
   342→          if (home) placeAt(home); // dropped outside — return without navigating
   343→        }
   344→      };
   345→
   346→      const onCancel = (ev: PointerEvent) => {
   347→        if (ev.pointerId !== gesture.pointerId) return;
   348→        abortGesture();
   349→      };
   350→
   351→      gesture = {
   352→        pointerId: e.pointerId,
   353→        startX: e.clientX,
   354→        startY: e.clientY,
   355→        grabOffset,
   356→        lastPos: axis === 'x' ? active.x : active.y,
   357→        dragging: false,
   358→        ul,
   359→        stop: () => {
   360→          window.removeEventListener('pointermove', onMove);
   361→          window.removeEventListener('pointerup', onUp);
   362→          window.removeEventListener('pointercancel', onCancel);
   363→        },
   364→      };
   365→      window.addEventListener('pointermove', onMove);
   366→      window.addEventListener('pointerup', onUp);
   367→      window.addEventListener('pointercancel', onCancel);
   368→      gestureRef.current = gesture;
   369→    },
   370→    [abortGesture, activeIndex, axis, onActivate, placeAt, x, y],
   371→  );
   372→
   373→  /** Unexpected capture loss mid-drag (element removed, browser takeover). */
   374→  const onLostCapture = useCallback(() => {
   375→    if (gestureRef.current?.dragging) abortGesture();
   376→  }, [abortGesture]);
   377→
   378→  /** After a real drag, the release click must not navigate a second time —
   379→   *  swallow exactly that pointer-initiated click. Keyboard activation
   380→   *  (detail === 0) always passes through untouched. */
   381→  const onClickCapture = useCallback(
   382→    (e: React.MouseEvent<HTMLUListElement>) => {
   383→      if (swallowClickRef.current && e.detail > 0) {
   384→        e.preventDefault();
   385→        e.stopPropagation();
   386→        swallowClickRef.current = false;
   387→      }
   388→    },
   389→    [],
   390→  );
   391→
   392→  // Unmounting mid-gesture: detach the window listeners.
   393→  useEffect(
   394→    () => () => {
   395→      gestureRef.current?.stop();
   396→      gestureRef.current = null;
   397→    },
   398→    [],
   399→  );
   400→
   401→  return {
   402→    itemRefs,
   403→    rect,
   404→    x,
   405→    y,
   406→    w,
   407→    h,
   408→    stretchX,
   409→    stretchY,
   410→    onPointerDown,
   411→    onLostCapture,
   412→    onClickCapture,
   413→  };
   414→}

   416→/** Roving focus inside a nav list: arrows move focus, Enter activates. */
   417→function handleNavKeys(e: React.KeyboardEvent<HTMLUListElement>) {
   418→  const buttons = Array.from(
   419→    e.currentTarget.querySelectorAll<HTMLButtonElement>('button[data-nav-item]')
   420→  );
   421→  if (buttons.length === 0) return;
   422→  const current = buttons.indexOf(document.activeElement as HTMLButtonElement);
   423→  let next = -1;
   424→  if (e.key === 'ArrowRight' || e.key === 'ArrowDown') next = (current + 1) % buttons.length;
   425→  else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') next = (current - 1 + buttons.length) % buttons.length;
   426→  else if (e.key === 'Home') next = 0;
   427→  else if (e.key === 'End') next = buttons.length - 1;
   428→  if (next >= 0) {
   429→    e.preventDefault();
   430→    buttons[next].focus();
   431→  }
   432→}

   504→function DesktopDock({
   505→  section,
   506→  onSectionChange,
   507→  dueCount,
   508→}: {
   509→  section: Section;
   510→  onSectionChange: (s: Section) => void;

   597→function FloatingTabBar({
   598→  section,
   599→  onSectionChange,
   600→  dueCount,
   601→}: {
   602→  section: Section;
   603→  onSectionChange: (s: Section) => void;
   604→  dueCount: number;
   605→}) {
   606→  const items = NAV_ITEMS.filter((n) => n.mobile);
   607→  // History is reachable on mobile (see-all in Listen) but absent from this
   608→  // bar — activeIndex is -1 then and the capsule hides instead of parking on
   609→  // the first tab.
   610→  const activeIndex = items.findIndex((n) => n.id === section);
   611→  const { itemRefs, rect, x, y, w, h, stretchX, stretchY, onPointerDown, onLostCapture, onClickCapture } =
   612→    useLiquidNav(items.length, activeIndex, 'x', onSectionChange);
   613→
   614→  return (
   615→    <nav
   616→      aria-label="Primary"
   617→      className="pointer-events-none fixed inset-x-0 bottom-0 z-30 flex justify-center pb-[calc(env(safe-area-inset-bottom)+12px)] lg:hidden"
   618→    >
   619→      <ul
   620→        onKeyDown={handleNavKeys}
   621→        onPointerDown={onPointerDown}
   622→        onLostPointerCapture={onLostCapture}
   623→        onClickCapture={onClickCapture}
   624→        className="df-nav-glass pointer-events-auto relative flex select-none items-center gap-0.5 rounded-[26px] border border-border/70 bg-card/72 p-1.5 shadow-[0_12px_36px_-10px_rgba(15,26,46,0.35)] backdrop-blur-2xl backdrop-saturate-150"
   625→      >
   626→        {rect && (
   627→          <motion.span
   628→            aria-hidden="true"
   629→            className="absolute left-0 top-0 rounded-[20px] border border-border/50 bg-accent shadow-[inset_0_1px_0_rgba(255,255,255,0.5)]"
   630→            style={{
   631→              x,
   632→              y,
   633→              width: w,
   634→              height: h,
   635→              scaleX: stretchX,
   636→              scaleY: stretchY,
   637→            }}
   638→          />
   639→        )}
   640→        {items.map((item, i) => {
   641→          const Icon = item.icon;
   642→          const active = section === item.id;
   643→          const showBadge = item.id === 'review' && dueCount > 0;
   644→          return (
   645→            <li key={item.id}>
   646→              <button
   647→                type="button"
   648→                data-nav-item
   649→                data-nav-id={item.id}
   650→                ref={(el) => {
   651→                  itemRefs.current[i] = el;
   652→                }}
   653→                onClick={() => onSectionChange(item.id)}
   654→                aria-current={active ? 'page' : undefined}
   655→                className={`relative flex h-[52px] w-[58px] touch-pan-y touch-pinch-zoom flex-col items-center justify-center gap-0.5 rounded-[20px] transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring active:scale-[0.94] ${
   656→                  active
   657→                    ? 'cursor-grab text-primary active:cursor-grabbing'
   658→                    : 'text-muted-foreground'
   659→                }`}
   660→              >
   661→                <span className="relative z-10 flex items-center">
   662→                  <Icon className="h-5 w-5" aria-hidden="true" />
   663→                  {showBadge && <NavBadge count={dueCount} />}
   664→                </span>
   665→                <span className="relative z-10 text-[10px] font-medium font-bangla">
   666→                  {item.bangla}
   667→                </span>
   668→              </button>
   669→            </li>
   670→          );
   671→        })}
   672→      </ul>
   673→    </nav>
   674→  );
